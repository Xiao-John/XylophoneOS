cmake_minimum_required(VERSION 3.21)
project(AuroraDesignPreview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
add_library(aurora_shell STATIC background.cpp docker.cpp applicationbrowser.cpp windowedge.cpp status.cpp)
target_link_libraries(aurora_shell PUBLIC Qt6::Widgets)
target_include_directories(aurora_shell PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(aurora_preview ../desktop_shell.cpp shell.qrc)
target_link_libraries(aurora_preview PRIVATE aurora_shell)
add_executable(aurora_shell_test shell_test.cpp shell.qrc)
target_link_libraries(aurora_shell_test PRIVATE aurora_shell Qt6::Test)
