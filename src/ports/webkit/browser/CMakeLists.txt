cmake_minimum_required(VERSION 3.21)
project(xos_clarke_viewport LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "XOS" OR NOT XOS_GLIBC_SDK)
    message(FATAL_ERROR "Use the matching XOS glibc Qt toolchain")
endif()
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Gui)
set(inputs "")
foreach(module IN ITEMS Core Gui BundledFreetype BundledHarfbuzz BundledPcre2 BundledZLIB)
    get_target_property(type Qt6::${module} TYPE)
    get_target_property(archive Qt6::${module} IMPORTED_LOCATION_RELEASE)
    file(REAL_PATH "${XOS_QT_PREFIX}/lib/libQt6${module}.a" expected)
    file(REAL_PATH "${archive}" selected)
    if(NOT type STREQUAL "STATIC_LIBRARY" OR NOT selected STREQUAL expected OR NOT EXISTS "${expected}")
        message(FATAL_ERROR "Viewport requires installed Qt6::${module} from XOS_QT_PREFIX")
    endif()
    string(APPEND inputs "${module}=${selected}\n")
endforeach()
string(APPEND inputs "moc=$<TARGET_FILE:Qt6::moc>\nrcc=$<TARGET_FILE:Qt6::rcc>\n")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/qt_inputs.txt" CONTENT "${inputs}")
qt6_add_executable(viewport viewport.cpp)
target_compile_features(viewport PRIVATE cxx_std_20)
target_link_libraries(viewport PRIVATE Qt6::Gui)
target_link_options(viewport PRIVATE "-Wl,--gc-sections,--no-keep-memory")
set_target_properties(viewport PROPERTIES SUFFIX ".elf")
include("${CMAKE_CURRENT_LIST_DIR}/../../qt6/glibc/xos_qpa.cmake")
xos_link_qpa(viewport "${XOS_QT_PREFIX}/share/fonts/DejaVuSans.ttf")
